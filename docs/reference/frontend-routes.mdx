```js
// Single source of truth for route classification, imported by both
// middleware.js (Edge runtime — keep this file free of Node/React APIs)
// and wrapper/Wrapper.js (client-side gating). Keeping one copy means a
// new public/auth-only page only needs to be added here, not kept in
// sync across two independent lists.

// Public: accessible by everyone (guest or authenticated), no redirect.
export const PUBLIC_PATHS = [
  "/",
  "/about-us",
  "/blogs",
  "/cookie-policy",
  "/for-job-seeker",
  "/for-mentor",
  "/for-recruiter",
  "/privacy-policy",
  "/terms-and-conditions",
  "/terms-of-use",
];

export const PUBLIC_PREFIXES = ["/blogs/", "/public-profile/"];

// Auth-only: guests only — authenticated users get bounced to /dashboard.
export const AUTH_ONLY_PATHS = ["/auth/sign-in", "/auth/sign-up", "/auth/forgot-password"];

export const AUTH_ONLY_PREFIXES = ["/auth/reset-password/"];

export function isPublicPath(pathname) {
  return PUBLIC_PATHS.includes(pathname) || PUBLIC_PREFIXES.some((p) => pathname.startsWith(p));
}

export function isAuthOnlyPath(pathname) {
  return (
    AUTH_ONLY_PATHS.includes(pathname) || AUTH_ONLY_PREFIXES.some((p) => pathname.startsWith(p))
  );
}

// Role-specific pages: the roles each one is for, matching the sidebar menus.
// Another role landing on one of these (typed URL, old link) is sent to the
// dashboard instead of a page whose API calls all fail. A rule covers the
// path itself and everything under it; the most specific rule wins. Pages
// not listed are open to every signed-in role.
const STUDENT = "STUDENT";
const COMPANY = "COMPANY";
const MENTOR = "MENTOR";

export const ROLE_ROUTES = {
  "/resume-builder": [STUDENT],
  "/roadmap": [STUDENT],
  "/mentorship": [STUDENT],
  "/jobs/ai-suggested-jobs": [STUDENT],
  "/myjobs": [STUDENT],

  "/add-new-job": [COMPANY],
  "/hired-candidates": [COMPANY],
  "/saved-searches": [COMPANY],
  "/myjobs/get-suggested-candidates": [COMPANY],

  "/candidates": [COMPANY, MENTOR],
  "/shortlistings": [COMPANY, MENTOR],

  "/mentor/services": [MENTOR],
  "/mentor/sessions": [MENTOR],
  "/mentor/availability": [MENTOR],
  "/mentor/start-webinar": [MENTOR],
};

// Exact-path rules: the job search page is for candidates, but /jobs/[id]
// (a single job) is also opened by the recruiter who posted it.
const ROLE_EXACT_ROUTES = {
  "/jobs": [STUDENT],
};

// The roles allowed on pathname, or null when any signed-in role may use it.
export function allowedRolesFor(pathname) {
  if (ROLE_EXACT_ROUTES[pathname]) return ROLE_EXACT_ROUTES[pathname];
  const match = Object.keys(ROLE_ROUTES)
    .filter((route) => pathname === route || pathname.startsWith(`${route}/`))
    .sort((a, b) => b.length - a.length)[0];
  return match ? ROLE_ROUTES[match] : null;
}

export function isRoleAllowed(pathname, role) {
  const roles = allowedRolesFor(pathname);
  return !roles || roles.includes(role);
}
```
